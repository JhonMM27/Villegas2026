CREATE TABLE IF NOT EXISTS cuadre_stocks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    fecha DATE NOT NULL,
    estado VARCHAR(20) DEFAULT 'completado',
    notas TEXT NULL,
    user_id SMALLINT UNSIGNED NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    INDEX idx_cuadre_stocks_fecha (fecha),
    INDEX idx_cuadre_stocks_estado (estado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
CREATE TABLE IF NOT EXISTS cuadre_stock_detalles (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    cuadre_stock_id BIGINT UNSIGNED NOT NULL,
    producto_id SMALLINT UNSIGNED NOT NULL,
    stock_sistema DECIMAL(10,4) NOT NULL,
    stock_fisico DECIMAL(10,4) NOT NULL,
    diferencia DECIMAL(10,4) NOT NULL,
    tipo VARCHAR(20) NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    CONSTRAINT fk_cuadre_stock_detalles_cuadre FOREIGN KEY (cuadre_stock_id) 
        REFERENCES cuadre_stocks(id) ON DELETE CASCADE,
    CONSTRAINT fk_cuadre_stock_detalles_producto FOREIGN KEY (producto_id) 
        REFERENCES productos(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
